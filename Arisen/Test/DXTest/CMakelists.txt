cmake_minimum_required(VERSION 3.29)

project(DXTest)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找 DXTest 源码
collect_sources(DXTEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src)

# 可执行目�?
add_executable(DXTest ${DXTEST_SOURCES})

# 设置预编译头
target_precompile_headers(DXTest PRIVATE src/stdafx.h)

target_include_directories(DXTest PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_include_directories(DXTest PRIVATE
    ${CMAKE_SOURCE_DIR}/Tools/Profiler
    ${CMAKE_SOURCE_DIR}/Core/Core.Foundation
    ${CMAKE_SOURCE_DIR}/Core/Core.Diagnostic
    ${CMAKE_SOURCE_DIR}/Core/Core.HAL
    ${CMAKE_SOURCE_DIR}/Engine/NativeEngine
    )  

# 设置�?Windows GUI 子系�?
set_target_properties(DXTest PROPERTIES
    WIN32_EXECUTABLE TRUE
)

target_link_libraries(DXTest PRIVATE Profiler Core.Foundation Core.Diagnostic Core.HAL RHI.DX12 NativeEngine)
target_link_libraries(DXTest PRIVATE
    d3d12
    dxgi
    d3dcompiler)
